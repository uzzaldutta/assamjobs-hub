async function main() {
  console.log("Seeding database...")
  for (const job of mockJobs) {
    const hash = crypto.createHash('sha256').update(job.title + job.organization).digest('hex')
    await prisma.job.upsert({
      where: { hash },
      update: {},
      create: {
        ...job,
        hash
      }
    })
  }
  console.log("Database seeded!")
}

main()
  .catch(e => {
    console.error(e)
    process.exit(1)
  })
  .finally(async () => {
    await prisma.$disconnect()
  })
